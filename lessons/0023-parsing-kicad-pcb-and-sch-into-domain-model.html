impl PcbLayer {
    pub fn from_kicad_name(name: &str) -> Result<Self, KicadLowerError> {
        match name {
            "B.Cu" => Ok(PcbLayer::BottomCopper),
            "F.Cu" => Ok(PcbLayer::),
            "F.SilkS" => Ok(PcbLayer::Silkscreen),
            "Ratsnest" => Ok(PcbLayer::Ratsnest),
            other => Err(KicadLowerError::(other.to_owned())),
        }
    }
}

}
